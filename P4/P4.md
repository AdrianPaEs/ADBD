## 2. Creación las tablas

** 1. Tabla Vivero **
```sql
CREATE TABLE vivero (
    id_vivero SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    latitud DECIMAL(9,6),
    longitud DECIMAL(9,6)
);
```

** 2. Tabla Empleados **
```sql
CREATE TABLE empleados (
    id_empleado SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    fecha_inicio DATE NOT NULL,
    fecha_fin DATE,
    id_vivero INT,
    FOREIGN KEY (id_vivero) REFERENCES vivero(id_vivero) ON DELETE SET NULL
);
```

** 3. Tabla Zona**
```sql
CREATE TABLE zona (
    id_zona SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    latitud DECIMAL(9,6),
    longitud DECIMAL(9,6),
    tarea VARCHAR(150),
    id_vivero INT NOT NULL,
    id_empleado INT,
    FOREIGN KEY (id_vivero) REFERENCES vivero(id_vivero) ON DELETE CASCADE,
    FOREIGN KEY (id_empleado) REFERENCES empleados(id_empleado) ON DELETE SET NULL
);
```

** 4. Tabla Producto **
```sql
CREATE TABLE producto (
    id_producto SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    precio DECIMAL(10,2) NOT NULL CHECK (precio >= 0)
);
```

** 5. Tabla Cliente Tajinaste Plus **
```sql
CREATE TABLE cliente_tajinaste_plus (
    id_cliente SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    fecha_ingreso DATE DEFAULT CURRENT_DATE NOT NULL
);
```

** 6. Tabla Pedidos**
```sql
CREATE TABLE pedidos (
    id_pedido SERIAL PRIMARY KEY,
    fecha DATE NOT NULL DEFAULT CURRENT_DATE,
    id_cliente INT NOT NULL,
    id_empleado INT,
    FOREIGN KEY (id_cliente) REFERENCES cliente_tajinaste_plus(id_cliente) ON DELETE CASCADE,
    FOREIGN KEY (id_empleado) REFERENCES empleados(id_empleado) ON DELETE SET NULL
);
```

** 7. Tabla Stock **
```sql
CREATE TABLE stock (
    id_zona INT,
    id_producto INT,
    cantidad INT NOT NULL CHECK (cantidad >= 0),
    PRIMARY KEY (id_zona, id_producto),
    FOREIGN KEY (id_zona) REFERENCES zona(id_zona) ON DELETE CASCADE,
    FOREIGN KEY (id_producto) REFERENCES producto(id_producto) ON DELETE CASCADE
);
```

** 8. Tabla Contiene **
```sql
CREATE TABLE contiene (
    id_producto INT,
    id_pedido INT,
    cantidad INT NOT NULL CHECK (cantidad > 0),
    PRIMARY KEY (id_producto, id_pedido),
    FOREIGN KEY (id_producto) REFERENCES producto(id_producto) ON DELETE CASCADE,
    FOREIGN KEY (id_pedido) REFERENCES pedidos(id_pedido) ON DELETE CASCADE
);
```

** 9. Tabla Bonificaciones **
```sql
CREATE TABLE bonificaciones (
    id_bonificacion SERIAL PRIMARY KEY,
    mes_anyo VARCHAR(7) NOT NULL, -- Formato 'MM-YYYY'
    id_cliente INT NOT NULL,
    FOREIGN KEY (id_cliente) REFERENCES cliente_tajinaste_plus(id_cliente) ON DELETE CASCADE
);
```

## 4. SELECT

** 1. En que vivero trabaja cada empleado**

SELECT e.nombre AS empleado, e.fecha_inicio, v.nombre AS vivero
FROM empleados e
LEFT JOIN vivero v ON e.id_vivero = v.id_vivero;

** 2. Stock de cada producto**
```sql
SELECT p.nombre AS producto, SUM(s.cantidad) AS stock_total
FROM producto p
JOIN stock s ON p.id_producto = s.id_producto
GROUP BY p.nombre, p.id_producto
ORDER BY stock_total DESC;
```

** 3. Historial de pedidos**
```sql
SELECT ped.id_pedido, ped.fecha, c.nombre AS cliente, e.nombre AS empleado_gestor
FROM pedidos ped
JOIN cliente_tajinaste_plus c ON ped.id_cliente = c.id_cliente
LEFT JOIN empleados e ON ped.id_empleado = e.id_empleado
ORDER BY ped.fecha DESC;
```
